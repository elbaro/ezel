Animation ​

To draw a multi-frame animated image, render each frame with MemoryCanvas, read its RGBA pixels, and add the frame to an animated WebP encoder.

rust
use std::{error::Error, f64::consts::TAU};

use ezel::{canvas::MemoryCanvas, dataflow::NumericSeries, prelude::*};
use webp_animation::{Encoder, EncoderOptions, EncodingConfig};

const FPS: usize = 60;
const SECONDS: usize = 10;
const NUM_FRAMES: usize = SECONDS * FPS;
const WIDTH: u32 = 800;
const HEIGHT: u32 = 600;
const POINT_COUNT: usize = 240;
const SAMPLES_PER_CYCLE: usize = 120;

fn sample(index: usize) -> f64 {
    (index as f64 * TAU / SAMPLES_PER_CYCLE as f64).sin()
}

fn update_plot(y: &NumericSeries, frame: usize) -> ezel::Result<()> {
    y.push(sample(POINT_COUNT + frame))?;
    y.truncate_front(1)?;
    Ok(())
}

fn main() -> Result<(), Box<dyn Error>> {
    let mut canvas = MemoryCanvas::new(WIDTH, HEIGHT)?;
    let graph = canvas.renderer_mut().graph().clone();

    let x = graph.numeric_input((0..POINT_COUNT).map(|index| index as f64).collect())?;
    let y = graph.numeric_input((0..POINT_COUNT).map(sample).collect::<Vec<_>>())?;

    let mut plot = Cartesian2::new().with_title("Animated Image");
    plot.add_geometry(ezel::cartesian2::Line::new(x, y.clone()).with_size(3.0));

    let mut encoder = Encoder::new_with_options(
        (WIDTH, HEIGHT),
        EncoderOptions {
            minimize_size: true,
            encoding_config: Some(EncodingConfig::new_lossy(80.0)),
            ..Default::default()
        },
    )?;

    for frame in 0..NUM_FRAMES {
        update_plot(&y, frame)?;
        canvas.render(&plot)?;
        let pixels = pollster::block_on(canvas.read_buffer())?;
        let timestamp_ms = i32::try_from(frame * 1_000 / FPS)?;
        encoder.add_frame(&pixels, timestamp_ms)?;
    }

    let animation = encoder.finalize(i32::try_from(SECONDS * 1_000)?)?;
    std::fs::write("anim.webp", &*animation)?;
    Ok(())
}
